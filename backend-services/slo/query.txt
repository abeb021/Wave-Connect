# Главное: эти выражения работают в vmui БЕЗ vmalert и recording rules.
# Копируй строку целиком (одной строкой) в поле запроса vmui.

# ---------------------------------------------------------------
# 1. SLI — RBSLO: доля запросов, которые НЕ завершились 5xx (5м)
#    Ожидаемо: около 1. Во время деградации feed падает к 0.
# ---------------------------------------------------------------
1 - sum by (service) (rate(wave_slo_requests_total{class="bad"}[5m])) / clamp_min(sum by (service) (rate(wave_slo_requests_total[5m])), 1e-9)

# ---------------------------------------------------------------
# 2. SLI — TBSLO: доля «хороших» минут (окно 5м)
#    Реагирует медленнее RBSLO — это и показываем в скринкасте.
# ---------------------------------------------------------------
sum by (service) (wave_slo_window_good{window="5m"}) / clamp_min(sum by (service) (wave_slo_window_total{window="5m"}), 1)

# ---------------------------------------------------------------
# 3. SLO — остаток error budget (0 = исчерпан, <0 = SLO нарушен)
#    Здесь окно 30d. Для короткого демо замени 30d на 10m.
# ---------------------------------------------------------------
1 - (1 - (1 - sum(increase(wave_slo_requests_total{service="feed",class="bad"}[30d])) / clamp_min(sum(increase(wave_slo_requests_total{service="feed"}[30d])), 1))) / 0.005

# ---------------------------------------------------------------
# 4. SLO — доля ошибок за час (0 = здорово, 0.1 = 10% ошибок)
#    Это же число удобно показывать на графике вместо remaining:
#    оно мгновенно реагирует и не требует 30 дней истории.
# ---------------------------------------------------------------
1 - sum(increase(wave_slo_requests_total{class="bad"}[1h])) / clamp_min(sum(increase(wave_slo_requests_total[1h])), 1)

# ---------------------------------------------------------------
# 5. SLA — доля успешных запросов по всему приложению (1h)
#    Плюс цель 0.99 второй строкой — видно, когда SLA нарушен.
# ---------------------------------------------------------------
1 - sum(increase(wave_slo_requests_total{class="bad"}[1h])) / clamp_min(sum(increase(wave_slo_requests_total[1h])), 1)

# ---------------------------------------------------------------
# 6. Причины 5xx: bad_gateway (сервис мёртв) vs server_error (сервис жив)
#    Ключевой график, доказывающий понимание.
# ---------------------------------------------------------------
sum by (service, reason) (rate(wave_slo_requests_total{class="bad"}[5m]))

# ---------------------------------------------------------------
# 7. Живость пробера (1 = проба успешна)
# ---------------------------------------------------------------
max by (service) (wave_slo_probe_up)

# ---------------------------------------------------------------
# 8. Burn rate — во сколько раз быстрее нормы горит бюджет (5м)
#    1 = ровно за окно, >14 = fast burn.
# ---------------------------------------------------------------
sum by (service) (rate(wave_slo_requests_total{class="bad"}[5m])) / clamp_min(sum by (service) (rate(wave_slo_requests_total[5m])), 1e-9) / 0.01
